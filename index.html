// Google Apps Script: bound to your Google Sheet (Extensions > Apps Script) // Reads the FIRST row as headers, returns the rest as JSON. // Supports JSONP: add ?callback=fn to the URL. function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; // first tab var values = sheet.getDataRange().getDisplayValues(); // strings, no Date objects var headers = values[0]; var data = values.slice(1) .filter(function (r) { return r.join('') !== ''; }) // skip blank rows .map(function (r) { var o = {}; headers.forEach(function (h, i) { o[h] = r[i]; }); return o; }); var json = JSON.stringify(data); var cb = e && e.parameter && e.parameter.callback; if (cb && /^[A-Za-z_$][\w$]*$/.test(cb)) { return ContentService.createTextOutput(cb + '(' + json + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); }
